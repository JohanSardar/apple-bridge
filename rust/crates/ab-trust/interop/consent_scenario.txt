func init() {
	if len(os.Args) > 1 && (os.Args[1] == "simapple" || os.Args[1] == "credential-cleanup") {
		if err := syscall.Exec(os.Getenv("AB_GO_LAB"), append([]string{os.Getenv("AB_GO_LAB")}, os.Args[1:]...), os.Environ()); err != nil {
			panic(err)
		}
	}
}
func TestRustConsentConnectorProduct(t *testing.T) {
	binary := os.Getenv("AB_CONSENT_FIXTURE")
	if binary == "" {
		t.Fatal("AB_CONSENT_FIXTURE required")
	}
	if err := os.Chdir(os.Getenv("APPLE_BRIDGE_LAB_ROOT")); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Minute)
	defer cancel()
	w, err := world.New(ctx, "sim", 1, 1, world.BlobOptions{Scope: "admin"})
	if err != nil {
		t.Fatal(err)
	}
	defer w.Close()
	a, err := trust.Open(w.Gateway.Home)
	if err != nil {
		t.Fatal(err)
	}
	store, err := state.Open(w.Gateway.Home)
	if err != nil {
		t.Fatal(err)
	}
	defer store.Close()
	if err = store.InitCommands(ctx); err != nil {
		t.Fatal(err)
	}
	backend := httptest.NewServer(http.HandlerFunc(func(out http.ResponseWriter, r *http.Request) {
		var input struct {
			ID      string          `json:"id"`
			Owner   trust.Principal `json:"owner"`
			Approve bool            `json:"approve"`
		}
		json.NewDecoder(r.Body).Decode(&input)
		var result any
		var err error
		switch r.URL.Path {
		case "/pending":
			var rows []contracts.CommandRecord
			rows, err = store.PendingApprovals(ctx)
			list := []map[string]any{}
			for _, row := range rows {
				redacted, _ := json.MarshalIndent(state.RedactedArguments(row.Command.Args), "", "  ")
				full, _ := json.MarshalIndent(row.Command.Args, "", "  ")
				list = append(list, map[string]any{"command_id": row.Command.ID, "operation": row.Command.Op, "principal": row.Principal, "arguments": string(redacted), "owner_arguments": string(full)})
			}
			result = list
		case "/command":
			result, err = store.Command(ctx, input.ID)
		case "/decision":
			token, _, e := a.CreateDeviceSession("trusted")
			if e != nil {
				err = e
				break
			}
			decision := "reject"
			if input.Approve {
				decision = "approve"
			}
			req, _ := http.NewRequestWithContext(ctx, "POST", w.Gateway.URL+"/approvals/"+input.ID, strings.NewReader("decision="+decision))
			req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
			req.Header.Set("Origin", w.Gateway.URL)
			req.AddCookie(&http.Cookie{Name: "ab_session_local", Value: token})
			var resp *http.Response
			resp, err = http.DefaultClient.Do(req)
			if err == nil {
				resp.Body.Close()
				if resp.StatusCode != 200 {
					err = fmt.Errorf("approval %d", resp.StatusCode)
				} else {
					result, err = store.Command(ctx, input.ID)
				}
			}
		default:
			err = fmt.Errorf("route")
		}
		if err != nil {
			http.Error(out, err.Error(), 500)
			return
		}
		json.NewEncoder(out).Encode(result)
	}))
	defer backend.Close()
	_, port, _ := net.SplitHostPort(w.Gateway.Host)
	command := exec.CommandContext(ctx, binary, w.Gateway.Home, "127.0.0.2:"+port, w.Gateway.URL, backend.URL)
	command.Stderr = os.Stderr
	stdout, err := command.StdoutPipe()
	if err != nil {
		t.Fatal(err)
	}
	if err = command.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() { command.Process.Kill(); command.Wait() }()
	scanner := bufio.NewScanner(stdout)
	if !scanner.Scan() {
		t.Fatal("fixture startup")
	}
	goOrigin := "http://localhost:" + port
	goClient := &webClient{home: w.Gateway.Home, ctx: ctx, origin: goOrigin}
	goAuthenticator, goErr := newWebAuthenticator()
	if goErr != nil {
		t.Fatal(goErr)
	}
	goEnroll, goErr := a.CreateWebEnroll()
	if goErr != nil {
		t.Fatal(goErr)
	}
	goStatus, goRaw, goErr := goClient.signIn(goAuthenticator, "enroll", goEnroll, "trusted")
	if goErr != nil || goStatus != 200 {
		t.Fatalf("Go enroll: %d %s %v", goStatus, goRaw, goErr)
	}
	old := http.DefaultTransport
	transport := old.(*http.Transport).Clone()
	dialer := net.Dialer{Timeout: 10 * time.Second}
	transport.DialContext = func(ctx context.Context, network, address string) (net.Conn, error) {
		if address == "localhost:"+port {
			address = "127.0.0.2:" + port
		}
		return dialer.DialContext(ctx, network, address)
	}
	http.DefaultTransport = transport
	defer func() { http.DefaultTransport = old; transport.CloseIdleConnections() }()
	origin := "http://localhost:" + port
	c := &webClient{home: w.Gateway.Home, ctx: ctx, origin: origin}
	goStatus, goRaw, err = c.signIn(goAuthenticator, "login", "", "trusted")
	if err != nil || goStatus != 200 {
		t.Fatalf("Rust login with Go-registered key: %d %s %v", goStatus, goRaw, err)
	}
	authenticator, err := newWebAuthenticator()
	if err != nil {
		t.Fatal(err)
	}
	enroll, err := a.CreateWebEnroll()
	if err != nil {
		t.Fatal(err)
	}
	status, raw, err := c.signIn(authenticator, "enroll", enroll, "trusted")
	if err != nil || status != 200 {
		t.Fatalf("Rust enroll: %d %s %v", status, raw, err)
	}
	http.DefaultTransport = old
	status, raw, err = c.signIn(authenticator, "login", "", "trusted")
	if err != nil || status != 200 {
		t.Fatalf("Go login with Rust-registered key: %d %s %v", status, raw, err)
	}
	http.DefaultTransport = transport
	status, raw, err = c.signIn(authenticator, "login", "", "trusted")
	if err != nil || status != 200 {
		t.Fatalf("Rust login: %d %s %v", status, raw, err)
	}
	e := Evidence{Scenario: "CONSENT-CONNECTOR", Level: "sim", Impl: "rust", Implementations: world.Implementations{Gateway: "go", Replica: "go"}, At: time.Now().UTC(), Latency: map[string]Latency{}, Metrics: map[string]any{"frontdoor": "rust", "gateway_backend": "go", "virtual_authenticator": true}}
	runConsent(ctx, w, c, &e, func(name string, ok bool) {
		e.Assertions = append(e.Assertions, Assertion{Name: name, Passed: ok})
		if !ok {
			t.Error(name)
		}
	})
	e.Passed = !t.Failed()
	encoded, _ := json.MarshalIndent(e, "", "  ")
	if path := os.Getenv("AB_CONSENT_EVIDENCE"); path != "" {
		os.WriteFile(path, encoded, 0600)
	}
	t.Logf("Rust frontdoor + Go gateway: %d assertions", len(e.Assertions))
}
